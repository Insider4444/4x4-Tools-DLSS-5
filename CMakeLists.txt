cmake_minimum_required(VERSION 3.24)
project(FourByFourToolsDLSS5 VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(DLSS5_STATIC_CRT ON)
include(cmake/CompilerFlags.cmake)
enable_testing()
add_executable(finishing_test tests/finishing_test.cpp src/host/finishing.cpp)
target_include_directories(finishing_test PRIVATE src/host)
apply_compiler_flags(finishing_test)
add_test(NAME finishing_controls COMMAND finishing_test)
set_tests_properties(finishing_controls PROPERTIES TIMEOUT 30)
option(BUILD_ADOBE_PLUGIN "Build the Adobe plug-in and GPU checks (requires local SDKs)" ON)
if(NOT BUILD_ADOBE_PLUGIN)
    return()
endif()
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The Adobe plug-in requires Windows x64 and MSVC.")
endif()
set(AE_SDK_ROOT "$ENV{AE_SDK_BASE_PATH}" CACHE PATH "Adobe SDK root containing Headers, Util, Resources")
set(NGX_SDK_ROOT "$ENV{DLSS_SDK_ROOT}" CACHE PATH "NVIDIA DLSS SDK root")
set(NR_RUNTIME_FILE "$ENV{DLSSNR_RUNTIME_DLL}" CACHE FILEPATH "Compatible, separately obtained nvngx_dlssnr.dll")
if(EXISTS "${AE_SDK_ROOT}/Examples/Headers/AE_Effect.h")
    set(AE_SDK_ROOT "${AE_SDK_ROOT}/Examples")
endif()
foreach(required "${AE_SDK_ROOT}/Headers/AE_Effect.h" "${AE_SDK_ROOT}/Util/entry.h"
    "${AE_SDK_ROOT}/Resources/PiPLtool.exe" "${NGX_SDK_ROOT}/include/nvsdk_ngx.h"
    "${NGX_SDK_ROOT}/lib/Windows_x86_64/x64/nvsdk_ngx_s.lib" "${NR_RUNTIME_FILE}")
    if(NOT EXISTS "${required}")
        message(FATAL_ERROR "Missing build dependency: ${required}. See docs/BUILDING.md.")
    endif()
endforeach()
enable_language(RC)
set(host_includes "${AE_SDK_ROOT}/Headers" "${AE_SDK_ROOT}/Headers/SP" "${AE_SDK_ROOT}/Util" "${CMAKE_SOURCE_DIR}/src/host")
set(pipl "${CMAKE_CURRENT_BINARY_DIR}/EffectPiPL")
add_custom_command(OUTPUT "${pipl}.rc"
    COMMAND "${CMAKE_CXX_COMPILER}" /nologo /P /TC /I "${AE_SDK_ROOT}/Headers" /I "${CMAKE_SOURCE_DIR}/src/host"
        "/Fi${pipl}.rr" "${CMAKE_SOURCE_DIR}/resources/VerifiedHostPiPL.r"
    COMMAND "${AE_SDK_ROOT}/Resources/PiPLtool.exe" "${pipl}.rr" "${pipl}.rrc"
    COMMAND "${CMAKE_CXX_COMPILER}" /nologo /P /TC /DMSWindows "/Fi${pipl}.rc" "${pipl}.rrc"
    DEPENDS resources/VerifiedHostPiPL.r src/host/effect_metadata.h VERBATIM)
add_library(DLSS5_Adobe MODULE src/host/adobe_effect.cpp src/host/neural_bridge.cpp src/host/finishing.cpp
    src/neural/Feature18Runtime.cpp "${pipl}.rc" resources/version.rc)
target_include_directories(DLSS5_Adobe PRIVATE ${host_includes} "${NGX_SDK_ROOT}/include" src/neural)
target_link_libraries(DLSS5_Adobe PRIVATE d3d12 dxgi version "${NGX_SDK_ROOT}/lib/Windows_x86_64/x64/nvsdk_ngx_s.lib")
set_target_properties(DLSS5_Adobe PROPERTIES PREFIX "" SUFFIX ".aex" OUTPUT_NAME "4x4Tools-DLSS5")
apply_compiler_flags(DLSS5_Adobe)
add_custom_command(TARGET DLSS5_Adobe POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E make_directory "$<TARGET_FILE_DIR:DLSS5_Adobe>/runtime"
    COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${NR_RUNTIME_FILE}" "$<TARGET_FILE_DIR:DLSS5_Adobe>/runtime/nvngx_dlssnr.dll" VERBATIM)
foreach(test host_contract neural_host)
    add_executable(${test}_test tests/${test}_test.cpp)
    target_include_directories(${test}_test PRIVATE ${host_includes})
    apply_compiler_flags(${test}_test)
endforeach()
add_test(NAME adobe_host_contract COMMAND host_contract_test "$<TARGET_FILE:DLSS5_Adobe>")
add_test(NAME adobe_neural_hardware COMMAND neural_host_test "$<TARGET_FILE:DLSS5_Adobe>")
set_tests_properties(adobe_host_contract PROPERTIES TIMEOUT 30)
set_tests_properties(adobe_neural_hardware PROPERTIES TIMEOUT 90)
add_executable(SupportCheck installer/support_check.cpp resources/version.rc)
target_compile_definitions(SupportCheck PRIVATE SUPPORT_CHECK)
target_include_directories(SupportCheck PRIVATE ${host_includes})
target_link_libraries(SupportCheck PRIVATE dxgi d3d12)
apply_compiler_flags(SupportCheck)
add_test(NAME installer_gpu_preflight COMMAND SupportCheck)
set_tests_properties(installer_gpu_preflight PROPERTIES TIMEOUT 60 WORKING_DIRECTORY "$<TARGET_FILE_DIR:DLSS5_Adobe>")
